g.vote(prop_id, 'did:veritas:carol', False, 3)
result = g.tally(prop_id, 100)
print(f'Result: {result}')
prop2 = g.create_proposal('did:veritas:alice', 'Elect arbiter', 'Elect bob as arbiter', ProposalType.ARBITER_ELECTION, {'arbiter_did': 'did:veritas:bob'})
g.vote(prop2, 'did:veritas:alice', True, 25)
g.tally(prop2, 100)
g.execute(prop2)
print(f'Is bob arbiter: {g.is_arbiter(\"did:veritas:bob\")}')
"
