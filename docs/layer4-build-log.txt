import json
from cryptography.hazmat.primitives.asymmetric import ed25519
with open('/tmp/did.json') as f: data = json.load(f)
did = data['did']
private_key = ed25519.Ed25519PrivateKey.from_private_bytes(bytes.fromhex(data['private_key']))
message = 'create:transaction:500:USDC'
signature = private_key.sign(message.encode()).hex()
import subprocess
result = subprocess.run(['curl', '-s', '-X', 'POST', 'http://localhost:8000/economic/transaction', '-H', 'Content-Type: application/json', '-d', json.dumps({'buyer_did': did, 'seller_did': 'did:veritas:bob', 'agent_id': 'agent_abc', 'amount': 500.0, 'currency': 'USDC', 'description': 'web_dev', 'signature': signature, 'message': message})], capture_output=True, text=True)
print(result.stdout)
"
