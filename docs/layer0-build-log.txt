nano core/engine.py
# PASTE: TrustEngine class
python3 -c "
from core.engine import TrustEngine
e = TrustEngine()
claim = e.submit_claim('user123', 'says hello', 'hello world')
print(f'Claim ID: {claim.id}')
verdict = e.evaluate(claim.id)
print(f'No proofs: {verdict.score.name} ({verdict.confidence})')
e.submit_proof(claim.id, 'hash_match', claim.content_hash, 'verifierA')
e.submit_proof(claim.id, 'signature', 'sig_abc123', 'verifierB')
e.submit_proof(claim.id, 'attestation', 'att_xyz789', 'verifierA')
verdict = e.evaluate(claim.id)
print(f'With proofs: {verdict.score.name} ({verdict.confidence}) — {verdict.explanation}')
"
